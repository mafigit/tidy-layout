package placement

// Reference cases for the Rust port tidy-layout.
// Not part of TALA: it only calls TALA's own functions and records what they answer.
//
// One move of one card, as the sized optimizer makes it: the places it looks at around a
// point (fillPlacementPointsGuarded), what the card costs at each of them
// (placementcost.NodeEdgeLength and NodeSymmetry) and the place it takes
// (moveNodeToBestGuarded).

import (
	"context"
	"encoding/json"
	"math/rand"
	"os"
	"testing"

	"github.com/d2lang/d2/d2layouts/d2talalayout/internal/layoutgraph"
	"github.com/d2lang/d2/d2layouts/d2talalayout/internal/limits"
	"github.com/d2lang/d2/d2layouts/d2talalayout/internal/placementcost"
	"github.com/d2lang/d2/lib/geo"
)

type moveInput struct {
	Name string `json:"name"`
	// 0: the cell size TALA computes from the card sizes.
	CellSize float64 `json:"cellSize"`
	// x, y, width, height
	Nodes [][4]float64 `json:"nodes"`
	// from, to (index into nodes), and the room the label needs (0, 0 without a label)
	Edges  [][4]int   `json:"edges"`
	Moving int        `json:"moving"`
	Point  [2]float64 `json:"point"`
}

type moveCase struct {
	moveInput
	Cell     float64 `json:"cell"`
	TurnCost float64 `json:"turnCost"`
	// The ring of the nearest free place; -1 when TALA finds none.
	Distance float64 `json:"distance"`
	// Per place TALA looks at, in its order: x, y, can the card go there (1/0), and if it can:
	// NodeEdgeLength and NodeSymmetry with the card there.
	Points [][5]float64 `json:"points"`
	// Where moveNodeToBestGuarded leaves the card.
	Best [2]float64 `json:"best"`
	// The same when the move must be an improvement.
	BestImproving [2]float64 `json:"bestImproving"`
}

func moveOf(t *testing.T, in moveInput) moveCase {
	ctx := context.Background()
	g := layoutgraph.NewGraph()
	nodes := make([]*layoutgraph.Node, 0, len(in.Nodes))
	for i, n := range in.Nodes {
		node := addTopLevel(g, layoutgraph.NewNode(layoutgraph.EntityID(i+1), n[2], n[3]))
		node.TopLeft = geo.NewPoint(n[0], n[1])
		nodes = append(nodes, node)
	}
	for _, e := range in.Edges {
		edge := g.Connect(nodes[e[0]], nodes[e[1]])
		edge.TargetArrowhead = layoutgraph.TriangleArrowhead
		if e[2] > 0 || e[3] > 0 {
			edge.MinWidth, edge.MinHeight = e[2], e[3]
			edge.Label = &layoutgraph.Label{Text: "label", Width: float64(e[2]), Height: float64(e[3])}
		}
	}
	g.ComputeCellSize()
	if in.CellSize != 0 {
		g.CellSize = in.CellSize
	}
	c := moveCase{moveInput: in, Cell: g.CellSize, TurnCost: g.TurnCost()}
	optim, err := newSizedOptimizer(ctx, g, nil, nil, rand.New(rand.NewSource(1)), nil)
	if err != nil {
		t.Fatal(err)
	}
	guard, err := limits.NewOptimizationWorkGuard(ctx, "LocalOptimize", limits.MaxOptimizationWorkUnits)
	if err != nil {
		t.Fatal(err)
	}
	if err := optim.rebuildSpatialIndex(guard); err != nil {
		t.Fatal(err)
	}
	moving := nodes[in.Moving]
	home := *moving.TopLeft
	p := geo.NewPoint(in.Point[0], in.Point[1])
	d, err := optim.findClosestUnoccupiedDistanceGuarded(moving, p, false, nil, guard)
	if err != nil {
		c.Distance = -1
		return c
	}
	c.Distance = d
	filled, err := optim.fillPlacementPointsGuarded(moving, p, d, false, &optim.placementScratch, guard)
	if err != nil {
		t.Fatal(err)
	}
	points := append([]geo.Point(nil), filled...)
	options := placementcost.EdgeLengthOptions{IncludeNodeSizes: true, EnforceMinimumGap: false, PenalizeDirection: true}
	for i := range points {
		point := points[i]
		can := point.X == home.X && point.Y == home.Y
		if !can {
			can, err = optim.indexedCanMove(moving, &point, guard)
			if err != nil {
				t.Fatal(err)
			}
		}
		row := [5]float64{point.X, point.Y, 0, 0, 0}
		if can {
			if err := optimizerMoveNodeAbs(moving, point.X, point.Y, guard); err != nil {
				t.Fatal(err)
			}
			length, err := placementcost.NodeEdgeLength(ctx, moving, options)
			if err != nil {
				t.Fatal(err)
			}
			symmetry, err := placementcost.NodeSymmetry(ctx, moving, nil)
			if err != nil {
				t.Fatal(err)
			}
			row[2], row[3], row[4] = 1, length, symmetry
			if err := optimizerMoveNodeAbs(moving, home.X, home.Y, guard); err != nil {
				t.Fatal(err)
			}
		}
		c.Points = append(c.Points, row)
	}
	best := func(mustImprove bool) [2]float64 {
		if _, err := optim.moveNodeToBestGuarded(ctx, moving, append([]geo.Point(nil), points...), mustImprove, guard); err != nil {
			t.Fatal(err)
		}
		at := [2]float64{moving.TopLeft.X, moving.TopLeft.Y}
		if err := optimizerMoveNodeAbs(moving, home.X, home.Y, guard); err != nil {
			t.Fatal(err)
		}
		return at
	}
	c.Best = best(false)
	c.BestImproving = best(true)
	if c.TurnCost != g.TurnCost() {
		t.Fatalf("the turn cost changed during the case")
	}
	return c
}

// moveCards makes a layout as a person leaves it. Tidy ones sit on a lattice, so that cards
// line up, mirror each other and stand in each other's way; the others lie anywhere on the
// editor's grid of 20.
func moveCards(r *rand.Rand, i int) moveInput {
	if i%3 == 2 {
		return mirrorCards(r)
	}
	in := moveInput{Edges: [][4]int{}}
	count := 2 + r.Intn(13)
	uniform := r.Intn(3) > 0
	tidy := r.Intn(2) == 0
	in.Name = "loose"
	if tidy {
		in.Name = "tidy"
	}
	for n := 0; n < count; n++ {
		w, h := 220.0, 120.0
		if !uniform {
			w = float64(60 + 20*r.Intn(14))
			h = float64(60 + 20*r.Intn(8))
		}
		x, y := float64(20*(r.Intn(70)-10)), float64(20*(r.Intn(45)-10))
		if tidy {
			x, y = float64(300*r.Intn(5)), float64(200*r.Intn(5))
			// Some a little out of line.
			if r.Intn(4) == 0 {
				x += float64(20 * (r.Intn(5) - 2))
				y += float64(20 * (r.Intn(5) - 2))
			}
		}
		in.Nodes = append(in.Nodes, [4]float64{x, y, w, h})
	}
	for e := 0; e < 1+r.Intn(count*2); e++ {
		a, b := r.Intn(count), r.Intn(count)
		if a == b {
			continue
		}
		// Now and then several connections between the same two cards.
		times := 1
		if r.Intn(6) == 0 {
			times = 2 + r.Intn(3)
		}
		for k := 0; k < times; k++ {
			edge := [4]int{a, b, 0, 0}
			if k > 0 && r.Intn(3) == 0 {
				edge[0], edge[1] = b, a
			}
			if r.Intn(2) == 0 {
				edge[2], edge[3] = 30+r.Intn(200), 20+r.Intn(30)
			}
			in.Edges = append(in.Edges, edge)
		}
	}
	in.Moving = r.Intn(count)
	// A card with connections tells more.
	for try := 0; try < 4; try++ {
		connected := false
		for _, e := range in.Edges {
			connected = connected || e[0] == in.Moving || e[1] == in.Moving
		}
		if connected {
			break
		}
		in.Moving = r.Intn(count)
	}
	switch r.Intn(3) {
	case 0:
		in.CellSize = 20
	case 1:
		in.CellSize = float64(40 * (1 + r.Intn(3)))
	}
	// Mostly dropped on or near another card, sometimes on its own place or anywhere.
	other := in.Nodes[r.Intn(count)]
	in.Point = [2]float64{other[0] + float64(r.Intn(160)-80), other[1] + float64(r.Intn(120)-60)}
	switch r.Intn(8) {
	case 0:
		own := in.Nodes[in.Moving]
		in.Point = [2]float64{own[0] + float64(r.Intn(30)-15), own[1] + float64(r.Intn(30)-15)}
	case 1:
		in.Point = [2]float64{float64(r.Intn(1500) - 200), float64(r.Intn(1000) - 200)}
	}
	return in
}

// mirrorCards drops a card where the cards it is connected to lie as mirror images of each
// other, exactly or nearly, with other cards around that may stand between them.
func mirrorCards(r *rand.Rand) moveInput {
	in := moveInput{Name: "mirror", Edges: [][4]int{}}
	w, h := 220.0, 120.0
	// A multiple of every cell size used here, so that the middle is a place on the grid.
	cx, cy := float64(440*(1+r.Intn(3))), float64(440*(1+r.Intn(2)))
	switch r.Intn(3) {
	case 0:
		in.CellSize = 20
	case 1:
		in.CellSize = 40
	}
	jitter := func() float64 {
		if r.Intn(3) > 0 {
			return 0
		}
		return float64(20 * (r.Intn(13) - 6))
	}
	// The moving card comes first: at the middle, near it, or far away.
	switch r.Intn(4) {
	case 0:
		in.Nodes = append(in.Nodes, [4]float64{cx, cy, w, h})
	case 1:
		in.Nodes = append(in.Nodes, [4]float64{cx + float64(20*(r.Intn(9)-4)), cy + float64(20*(r.Intn(9)-4)), w, h})
	default:
		in.Nodes = append(in.Nodes, [4]float64{float64(20 * r.Intn(90)), float64(20 * r.Intn(60)), w, h})
	}
	connect := func(a, b int) {
		edge := [4]int{a, b, 0, 0}
		if r.Intn(2) == 0 {
			edge[0], edge[1] = b, a
		}
		if r.Intn(3) == 0 {
			edge[2], edge[3] = 30+r.Intn(120), 20+r.Intn(30)
		}
		in.Edges = append(in.Edges, edge)
	}
	for pair := 0; pair < 1+r.Intn(3); pair++ {
		nw, nh := w, h
		if r.Intn(4) == 0 {
			nw, nh = float64(100+20*r.Intn(12)), float64(60+20*r.Intn(8))
		}
		gap := float64(60 + 20*r.Intn(12))
		var a, b [4]float64
		if r.Intn(2) == 0 {
			y := cy + (h-nh)/2
			a = [4]float64{cx - gap - nw + jitter(), y + jitter(), nw, nh}
			b = [4]float64{cx + w + gap + jitter(), y + jitter(), nw, nh}
		} else {
			x := cx + (w-nw)/2
			a = [4]float64{x + jitter(), cy - gap - nh + jitter(), nw, nh}
			b = [4]float64{x + jitter(), cy + h + gap + jitter(), nw, nh}
		}
		// Now and then one of the two is of another size.
		if r.Intn(6) == 0 {
			b[2], b[3] = float64(60+20*r.Intn(20)), float64(60+20*r.Intn(12))
		}
		in.Nodes = append(in.Nodes, a, b)
		connect(0, len(in.Nodes)-2)
		connect(0, len(in.Nodes)-1)
		// A card beyond one of the two, so that this one has a mirror pair of its own.
		if r.Intn(3) == 0 {
			in.Nodes = append(in.Nodes, [4]float64{2*a[0] - cx + jitter(), 2*a[1] - cy + jitter(), w, h})
			connect(len(in.Nodes)-3, len(in.Nodes)-1)
		}
	}
	for extra := 0; extra < r.Intn(6); extra++ {
		in.Nodes = append(in.Nodes, [4]float64{cx + float64(20*(r.Intn(70)-35)), cy + float64(20*(r.Intn(50)-25)), float64(60 + 20*r.Intn(12)), float64(60 + 20*r.Intn(6))})
		if r.Intn(2) == 0 {
			connect(r.Intn(len(in.Nodes)-1), len(in.Nodes)-1)
		}
	}
	// Dropped on the middle, or a little beside it.
	in.Point = [2]float64{cx + float64(r.Intn(15)-7), cy + float64(r.Intn(15)-7)}
	if r.Intn(3) == 0 {
		in.Point[0] += float64(20 * (r.Intn(7) - 3))
		in.Point[1] += float64(20 * (r.Intn(7) - 3))
	}
	return in
}

func TestExportMoveReference(t *testing.T) {
	out := os.Getenv("MOVE_REFERENCE_OUT")
	if out == "" {
		t.Skip("set MOVE_REFERENCE_OUT to write the reference cases")
	}
	r := rand.New(rand.NewSource(20261122))
	var cases []moveCase
	points := 0
	for i := 0; len(cases) < 300; i++ {
		c := moveOf(t, moveCards(r, i))
		// Keep the file small: a card dropped far inside a heap of cards looks at thousands of places.
		if len(c.Points) > 400 {
			continue
		}
		points += len(c.Points)
		cases = append(cases, c)
	}
	if found := os.Getenv("REFERENCE_FOUND"); found != "" {
		// Layouts found by the probe (docs/reference.md, "Finding cases").
		data, err := os.ReadFile(found)
		if err != nil {
			t.Fatal(err)
		}
		var extra []moveInput
		if err := json.Unmarshal(data, &extra); err != nil {
			t.Fatal(err)
		}
		for _, in := range extra {
			in.Name = "found"
			c := moveOf(t, in)
			points += len(c.Points)
			cases = append(cases, c)
		}
	}
	t.Logf("%d cases, %d places", len(cases), points)
	data, err := json.Marshal(cases)
	if err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(out, data, 0o644); err != nil {
		t.Fatal(err)
	}
}

// TestMoveProbe answers layouts made elsewhere: PROBE_IN holds a list of inputs, PROBE_OUT gets
// TALA's answers in the same order.
func TestMoveProbe(t *testing.T) {
	in, out := os.Getenv("PROBE_IN"), os.Getenv("PROBE_OUT")
	if in == "" || out == "" {
		t.Skip("set PROBE_IN and PROBE_OUT")
	}
	data, err := os.ReadFile(in)
	if err != nil {
		t.Fatal(err)
	}
	var inputs []moveInput
	if err := json.Unmarshal(data, &inputs); err != nil {
		t.Fatal(err)
	}
	answers := make([]moveCase, 0, len(inputs))
	for _, input := range inputs {
		answers = append(answers, moveOf(t, input))
	}
	result, err := json.Marshal(answers)
	if err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(out, result, 0o644); err != nil {
		t.Fatal(err)
	}
}
